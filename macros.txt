Sub Clear_POL_Slicer()

    Dim slcrCache As SlicerCache
    Dim slcr As Slicer
    Dim targetSheetName As String
    Dim pt As PivotTable
    Dim pf As PivotField
    
    targetSheetName = "POL_Booking_Flash"
    
    ' 1. Optimization: Turn off updates at the very start of the macro
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    
    ' Set table references early to pause layout updates safely
    On Error Resume Next
    Set pt = Sheets(targetSheetName).PivotTables("PivotTable2")
    Set pf = pt.PivotFields("Status ")
    If Not pt Is Nothing Then pt.ManualUpdate = True
    On Error GoTo 0
    
    ' 2. Loop through and clear only the slicers on the target sheet
    For Each slcrCache In ActiveWorkbook.SlicerCaches
        For Each slcr In slcrCache.Slicers
            If slcr.Shape.Parent.Name = targetSheetName Then
                slcrCache.ClearManualFilter
                Exit For ' Move to next cache once a match is cleared
            End If
        Next slcr
    Next slcrCache
    
    ' 3. Safely clear the specific Pivot Field filter
    If Not pf Is Nothing Then
        On Error Resume Next
        pf.ClearAllFilters
        On Error GoTo 0
    End If
    
    ' 4. Clean up, re-enable updates, and recalculate everything
    If Not pt Is Nothing Then pt.ManualUpdate = False
    
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True


End Sub

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim cell As Range
    Dim keywords As Variant, keyword As Variant
    Dim startPos As Long, endPos As Long, pos As Long
    Dim lastRowAI As Long, lastRow As Long, lastRowAG As Long
    Dim ws As Worksheet
    Dim dotPos As Long
    Dim cellText As String
    Dim targetRange As Range

    ' Only run if the change happens in the relevant data columns
    If Intersect(Target, Me.Range("A:AL")) Is Nothing Then Exit Sub

    On Error GoTo CleanExit

    Set ws = Me

    ' Performance Boost & Loop Prevention
    Application.ScreenUpdating = False
    Application.EnableEvents = False

    ' --- PART 0: Timestamps ---
    'On Error Resume Next
    'Worksheets("POL_Booking_Flash").Range("W1").Value = Now
    'ws.Range("N1").Value = Now
    'On Error GoTo CleanExit

    ' --- PART 1: Remove Decimals ---
    lastRowAI = ws.Cells(ws.Rows.Count, "AI").End(xlUp).Row
    If lastRowAI >= 13 Then
        For Each cell In ws.Range("AI13:AJ" & lastRowAI)
            If Not IsError(cell.Value) And cell.Value <> "" Then
                cellText = cell.Value
                Do While InStr(cellText, ".") > 0
                    dotPos = InStr(cellText, ".")
                    cellText = Left(cellText, dotPos - 1) & Mid(cellText, dotPos + 3)
                Loop
                If cell.Value <> cellText Then cell.Value = IIf(Trim(cellText) = "", " ", cellText)
            End If
        Next cell
    End If

    ' --- PART 2: Bold Keywords ---
    keywords = Array("AEJEA:", "CNNGB:", "CNSHA:", "CNSHK:", "CR/GT/NI Discharge:", _
                     "EB PA Load:", "FRFOS/ESVLC Discharge:", "GRPIR:", "INPAV:", _
                     "KRPUS:", "LATAM Load:", "MX Laden:", "PA Load:", _
                     "PAROD Discharge :", "PAROD Move Count Unit:", "LKCMB:", "SGSIN:", _
                     "BEANR:", "DEBRV:", "DEHAM:", "FRLEH:", "GBSOU:", "LATAM:", "NLRTM:", "THLCH:", "VNCMP:", _
                     "ESALG:", "ESVLC:", "FRFOS:", "ITGOA:", "ITSPE:", _
                     "Laden:", _
                     "ITSAL:", "TRALI:", "TRISK:", "TRIST:", "INNSA:", "MTY:", "PAROD Discharge:")

    If lastRowAI >= 13 Then
        For Each cell In ws.Range("AI13:AJ" & lastRowAI)
            If Not IsError(cell.Value) And cell.Value <> "" Then
                cell.Font.Bold = False
                cell.Font.ColorIndex = xlAutomatic
                For Each keyword In keywords
                    startPos = InStr(1, cell.Value, keyword, vbTextCompare)
                    Do While startPos > 0
                        cell.Characters(startPos, Len(keyword)).Font.Bold = True
                        startPos = InStr(startPos + Len(keyword), cell.Value, keyword, vbTextCompare)
                    Loop
                Next keyword
            End If
        Next cell
    End If

    ' --- PART 3: Red text between + and * ---
' --- PART 3: Red text between +###* and (-###*) ---

Dim re As Object
Dim matches As Object
Dim m As Object

Set re = CreateObject("VBScript.RegExp")

For Each cell In ws.Range("AI13:AJ" & lastRowAI)

    If Not IsError(cell.Value) And Len(cell.Value) > 0 Then

        cellText = CStr(cell.Value)

        ' Positive values: +123*
        With re
    .Global = True
    .Pattern = "\+\d+\*+"
End With

Set matches = re.Execute(cellText)

For Each m In matches
    cell.Characters(m.FirstIndex + 1, m.Length).Font.Color = RGB(255, 0, 0)
Next m

' Negative values: (-123*), (-123**), (-123***)
With re
    .Global = True
    .Pattern = "\(-\d+\*+\)"
End With

Set matches = re.Execute(cellText)

For Each m In matches
    cell.Characters(m.FirstIndex + 1, m.Length).Font.Color = RGB(255, 0, 0)
Next m

    End If

Next cell


    ' --- PART 4: Hide * Cells (FIXED) ---
    ' Using AG to find the last row to ensure we cover all data rows
    lastRowAG = ws.Cells(ws.Rows.Count, "AH").End(xlUp).Row
    If lastRowAG >= 13 Then
        Set targetRange = ws.Range("AH13:AN" & lastRowAG)
        For Each cell In targetRange
            If Not IsError(cell.Value) Then
                ' FIX: Only hide if the cell contains ONLY "*" or starts with "*"
                ' This prevents hiding actual descriptive text that might contain an asterisk
                If Trim(cell.Text) = "*" Then
                    cell.NumberFormat = ";;;"
                Else
                    ' Reset to General only if it was hidden, otherwise keep current format
                    If cell.NumberFormat = ";;;" Then cell.NumberFormat = "General"
                End If
            End If
        Next cell
    End If


    ' --- PART 5: Copy Formatting ---
    ws.Range("AH12").Copy
    ws.Range("AI12").PasteSpecial Paste:=xlPasteFormats
    Application.CutCopyMode = False

    With ws.Range("AH12:AJ12")
       .HorizontalAlignment = xlCenter
       .VerticalAlignment = xlCenter
       .WrapText = False
    End With
    
  ' This macro automatically fires every time a Slicer is clicked
    On Error Resume Next
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    
    ' Explicitly freeze the row heights right after the slicer updates
    Me.Range("13:1000").RowHeight = 15
    
    
    


CleanExit:
    Application.EnableEvents = True
    Application.ScreenUpdating = True
End Sub

Private Sub ToggleColumns(ColRange As String, ShownText As String, HiddenText As String)
    Dim xRange As Range
    Dim shp As Object
    
    ' Set the target columns
    Set xRange = ActiveSheet.Columns(ColRange)
    
    ' Identify which shape was clicked
    On Error Resume Next
    Set shp = ActiveSheet.Shapes(Application.Caller)
    On Error GoTo 0

    If Not shp Is Nothing Then
        ' Force solid fill to prevent "half-green" gradient glitches
        shp.Fill.Solid
        ' Keep text bold in both states
        shp.TextFrame.Characters.Font.Bold = True
        
        ' Toggle Logic: If columns are hidden, show them. Otherwise, hide them.
        If xRange.EntireColumn.Hidden = True Then
            ' STATE: SHOW COLUMNS
            xRange.EntireColumn.Hidden = False
            shp.TextFrame.Characters.Text = ShownText
            shp.Fill.ForeColor.RGB = RGB(146, 208, 80) ' Green
        Else
            ' STATE: HIDE COLUMNS
            xRange.EntireColumn.Hidden = True
            shp.TextFrame.Characters.Text = HiddenText
            shp.Fill.ForeColor.RGB = RGB(225, 225, 225) ' Grey
        End If
        
        ' Standardize Alignment
        shp.TextFrame.HorizontalAlignment = -4108 ' Center
        shp.TextFrame.VerticalAlignment = -4108   ' Center
    End If
    
    ActiveSheet.Range("S2").Select
End Sub

Sub Apply_Formatting()

    Dim cell As Range
    Dim keywords As Variant, keyword As Variant
    Dim startPos As Long, endPos As Long, pos As Long
    Dim lastRowAI As Long, lastRow As Long, lastRowAG As Long
    Dim ws As Worksheet
    Dim dotPos As Long
    Dim cellText As String
    Dim targetRange As Range
    Dim re As Object
    Dim matches As Object
    Dim m As Object

    On Error GoTo CleanExit

    ' Set ws to the ActiveSheet since this runs from a standard module
    Set ws = ActiveSheet

    ' Performance Boost & Loop Prevention
    Application.ScreenUpdating = False
    Application.EnableEvents = False

    ' --- PART 1: Remove Decimals ---
    lastRowAI = ws.Cells(ws.Rows.Count, "AI").End(xlUp).Row
    If lastRowAI >= 13 Then
        For Each cell In ws.Range("AI13:AJ" & lastRowAI)
            If Not IsError(cell.Value) And cell.Value <> "" Then
                cellText = cell.Value
                Do While InStr(cellText, ".") > 0
                    dotPos = InStr(cellText, ".")
                    cellText = Left(cellText, dotPos - 1) & Mid(cellText, dotPos + 3)
                Loop
                If cell.Value <> cellText Then cell.Value = IIf(Trim(cellText) = "", " ", cellText)
            End If
        Next cell
    End If

    ' --- PART 2: Bold Keywords ---
    keywords = Array("AEJEA:", "CNNGB:", "CNSHA:", "CNSHK:", "CR/GT/NI Discharge:", _
                     "EB PA Load:", "FRFOS/ESVLC Discharge:", "GRPIR:", "INPAV:", _
                     "KRPUS:", "LATAM Load:", "MX Laden:", "PA Load:", _
                     "PAROD Discharge :", "PAROD Move Count Unit:", "LKCMB:", "SGSIN:", _
                     "BEANR:", "DEBRV:", "DEHAM:", "FRLEH:", "GBSOU:", "LATAM:", "NLRTM:", "THLCH:", "VNCMP:", _
                     "ESALG:", "ESVLC:", "FRFOS:", "ITGOA:", "ITSPE:", _
                     "Laden:", _
                     "ITSAL:", "TRALI:", "TRISK:", "TRIST:", "INNSA:", "MTY:", "PAROD Discharge:")

    If lastRowAI >= 13 Then
        For Each cell In ws.Range("AI13:AJ" & lastRowAI)
            If Not IsError(cell.Value) And cell.Value <> "" Then
                cell.Font.Bold = False
                cell.Font.ColorIndex = xlAutomatic
                For Each keyword In keywords
                    startPos = InStr(1, cell.Value, keyword, vbTextCompare)
                    Do While startPos > 0
                        cell.Characters(startPos, Len(keyword)).Font.Bold = True
                        startPos = InStr(startPos + Len(keyword), cell.Value, keyword, vbTextCompare)
                    Loop
                Next keyword
            End If
        Next cell
    End If
    
  ' --- PART 3: Red text between +###* and (-###*) ---
    Set re = CreateObject("VBScript.RegExp")

    If lastRowAI >= 13 Then
        For Each cell In ws.Range("AI13:AJ" & lastRowAI)
            If Not IsError(cell.Value) And Len(cell.Value) > 0 Then
                cellText = CStr(cell.Value)
        
                ' Positive values: +123*
                With re
                    .Global = True
                    .Pattern = "\+\d+\*+"
                End With
        
                Set matches = re.Execute(cellText)
                For Each m In matches
                    cell.Characters(m.FirstIndex + 1, m.Length).Font.Color = RGB(255, 0, 0)
                Next m
        
                ' Negative values: (-123*), (-123**), (-123***)
                With re
                    .Global = True
                    .Pattern = "\(-\d+\*+\)"
                End With
        
                Set matches = re.Execute(cellText)
                For Each m In matches
                    cell.Characters(m.FirstIndex + 1, m.Length).Font.Color = RGB(255, 0, 0)
                Next m
            End If
        Next cell
    End If

    ' --- PART 4: Hide * Cells ---
    lastRowAG = ws.Cells(ws.Rows.Count, "AH").End(xlUp).Row
    If lastRowAG >= 13 Then
        Set targetRange = ws.Range("AH13:AN" & lastRowAG)
        For Each cell In targetRange
            If Not IsError(cell.Value) Then
                If Trim(cell.Text) = "*" Then
                    cell.NumberFormat = ";;;"
                Else
                    If cell.NumberFormat = ";;;" Then cell.NumberFormat = "General"
                End If
            End If
        Next cell
    End If

    ' --- PART 5: Copy Formatting ---
    ws.Range("AH12").Copy
    ws.Range("AI12").PasteSpecial Paste:=xlPasteFormats
    Application.CutCopyMode = False

    With ws.Range("AH12:AJ12")
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .WrapText = False
    End With
    
    ' Explicitly freeze the row heights
    ws.Range("13:1000").RowHeight = 15

CleanExit:
    ' Restore settings safely
    Application.EnableEvents = True
    Application.ScreenUpdating = True
End Sub
